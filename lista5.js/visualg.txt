Exercicio 3:
algoritmo "Soma_ate_Zero"

var
   n1, i, soma: inteiro

inicio

   escreva("Insira um numero: ")
   leia(n1)

   soma <- 0
   i <- n1

   enquanto i >= 0 faca
      soma <- soma + i
      escreval("A soma atual e de: ", i)
      escreval("Resultado acumulado: ", soma)
      i <- i - 1
   fimenquanto

fimalgoritmo

Exercicio 5:
algoritmo "Media_Notas"

var
   nota, soma, media: real
   quantidade: inteiro

inicio

   soma <- 0
   quantidade <- 0

   escreva("Digite uma nota (-1 para encerrar): ")
   leia(nota)

   enquanto nota <> -1 faca

      soma <- soma + nota
      quantidade <- quantidade + 1

      escreva("Digite uma nota (-1 para encerrar): ")
      leia(nota)

   fimenquanto

   media <- soma / quantidade

   escreval("Media: ", media)

fimalgoritmo


Exercicio 6:
algoritmo "Tabuada"

var
   n1, i, multiplicacao: inteiro

inicio

   escreva("Insira um numero: ")
   leia(n1)

   se n1 = 0 entao
      escreval("Numero invalido.")
   senao
      para i de 0 ate 10 faca
         multiplicacao <- n1 * i
         escreval(n1, " x ", i, " = ", multiplicacao)
      fimpara
   fimse

fimalgoritmo
